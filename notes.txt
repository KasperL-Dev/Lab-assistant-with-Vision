########## Overlay items ##########
# The overlay is a list of items drawn in the upper-left corner, top to bottom.
# Each item is a dict with:
#   "label" : str   – the text shown, e.g. "FPS" or "Robot"
#   "value" : callable or str
#               • callable  → called every frame with no arguments; return value is shown
#               • plain str → shown as-is (static text)
#   "color" : (B, G, R) tuple, default (255, 255, 255)
#
# ── HOW TO ADD AN ITEM ───────────────────────────────────────────────────────
# 1. Static text:
#       overlay_items.append({ "label": "Mode", "value": "Auto", "color": (200, 200, 200) })
#
# 2. Live value from a variable – wrap it in a lambda:
#       robot_status = "IDLE"
#       overlay_items.append({ "label": "Robot", "value": lambda: robot_status })
#
# 3. Live value from a function you already have:
#       overlay_items.append({ "label": "Queue", "value": get_queue_length })
#
# Colors are BGR, e.g. green=(0,255,0), yellow=(0,255,255), red=(0,0,255).
# ─────────────────────────────────────────────────────────────────────────────
